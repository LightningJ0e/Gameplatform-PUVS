
# Zur installation von kubectl und minikube
winget install Kubernetes.kubectl
winget install Kubernetes.minikube

minikube start --driver=docker --cpus=4 --memory=8192
minikube kubectl -- get nodes
minikube kubectl -- -n kube-system get pods
cd ./pg-demo
minikube kubectl -- apply -f .\postgres-k8s.yaml
minikube kubectl -- apply -f .\adminer-k8s.yaml

ONLY FIRST TIME: minikube kubectl -- -n postgresdb exec -it postgres-0 -- psql -U game_db_admin -d postgres -c "create database game_db;"

In one shell: minikube kubectl -- -n postgresdb get pods -l app=adminer -w

in the other shell: minikube service adminer -n postgresdb --url


When starting normally:
minikube start --driver=docker --cpus=4 --memory=8192

In one shell: minikube kubectl -- -n postgresdb get pods -l app=adminer -w

in the other shell: minikube service adminer -n postgresdb --url

in a third shell: minikube kubectl -- -n postgresdb port-forward svc/postgres 55432:5432

in fourth shell: uvicorn Services.<YourService>.main:app --reload
then open: http://127.0.0.1:8000/docs
